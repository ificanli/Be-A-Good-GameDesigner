            function Result=Markov(P,x,y,c)

            %Result=[upgrade_time,total_cost]
            % 1upgrade_timexy
            % 2total_cost

            %P
            [P_size,~]=size(P);
            %
            I=eye(P_size);

            Iy=zeros(P_size,1);Iy(y)=1;
            Ey=I;Ey(y,y)=0;
            %
            if x==y
            upgrade_time=0;
            total_cost=0;
            else
            upgrade_time=Ix*(I-P*Ey)^-2*P*Iy;
            total_cost=Ix*((I-P*Ey)^-1*P+I)*Ey*c';
            end
            Result=[upgrade_time,total_cost];
